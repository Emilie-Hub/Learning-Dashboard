// Google Apps Script (Code.gs) - 快樂森林學堂 SEL 雲端雙向同步腳本
function doGet(e) {
  var action = e.parameter.action;
  if (action === 'fetch') {
    return handleFetch();
  }
  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "SEL 雲端服務運作中" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    
    if (action === 'upload') {
      return handleUpload(contents);
    } else if (action === 'fetch') {
      return handleFetch();
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "未知的操作指令" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function handleUpload(payload) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetStudents = getOrCreateSheet(ss, "即時隊員名冊");
  sheetStudents.clear();
  sheetStudents.appendRow(["學生ID", "姓氏", "名字", "動物頭像", "目前星星", "累積星星", "今日簽到", "心情天氣", "簽到時間", "共讀印章", "任務印章", "整潔印章", "自訂照片", "最後更新"]);
  
  if (payload.students && payload.students.length > 0) {
    payload.students.forEach(function(s) {
      sheetStudents.appendRow([
        s.id, s.surname, s.name, s.animalType, s.stars || 0, s.lifetimeStars || 0,
        s.checkedIn ? "是" : "否", s.mood || "", s.time || "",
        s.tasks && s.tasks.reading ? "V" : "-",
        s.tasks && s.tasks.homework ? "V" : "-",
        s.tasks && s.tasks.tidy ? "V" : "-",
        s.customAvatar ? "有 (Base64)" : "預設頭像",
        payload.timestamp || new Date().toISOString()
      ]);
    });
  }

  var sheetSnapshot = getOrCreateSheet(ss, "系統備份快照");
  sheetSnapshot.clear();
  sheetSnapshot.appendRow(["快照時間", "JSON備份資料"]);
  sheetSnapshot.appendRow([payload.timestamp || new Date().toISOString(), JSON.stringify(payload)]);

  return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "雲端備份成功！" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function handleFetch() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetSnapshot = ss.getSheetByName("系統備份快照");
  
  if (!sheetSnapshot || sheetSnapshot.getLastRow() < 2) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "雲端尚無備份快照" }))
      .setMimeType(ContentService.MimeType.JSON);
  }

  var rawJson = sheetSnapshot.getRange(2, 2).getValue();
  var backupData = JSON.parse(rawJson);

  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: backupData }))
    .setMimeType(ContentService.MimeType.JSON);
}

function getOrCreateSheet(ss, name) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
  }
  return sheet;
}
